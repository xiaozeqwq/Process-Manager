public class ChartDataTracker {
    private static final int MAX_POINTS = 60;
    
    private final LinkedList<Double> downloadSpeeds = new LinkedList<>();
    private final LinkedList<Double> uploadSpeeds = new LinkedList<>();
    private final LinkedList<Double> cpuUsages = new LinkedList<>();
    private final LinkedList<Double> memoryUsages = new LinkedList<>();

    /**
     * æ·»åŠ ä¸€ç»„æ–°çš„æ•°æ®ç‚¹
     * @param download ä¸‹è½½é€Ÿåº¦ (KB/s)
     * @param upload ä¸Šä¼ é€Ÿåº¦ (KB/s)
     * @param cpu CPU å ç”¨ç‡ (%)
     * @param memory å†…å­˜å ç”¨ (MB)
     */
    public synchronized void addDataPoint(double download, double upload, double cpu, double memory) {
        addToQueue(downloadSpeeds, download);
        addToQueue(uploadSpeeds, upload);
        addToQueue(cpuUsages, cpu);
        addToQueue(memoryUsages, memory);
    }

    private void addToQueue(LinkedList<Double> queue, double value) {
        queue.add(value);
        if (queue.size() > MAX_POINTS) {
            queue.removeFirst();
        }
    }

    public synchronized LinkedList<Double> getDownloadSpeeds() { return new LinkedList<>(downloadSpeeds); }
    public synchronized LinkedList<Double> getUploadSpeeds() { return new LinkedList<>(uploadSpeeds); }
    public synchronized LinkedList<Double> getCpuUsages() { return new LinkedList<>(cpuUsages); }
    public synchronized LinkedList<Double> getMemoryUsages() { return new LinkedList<>(memoryUsages); }
    
    public synchronized boolean isEmpty() {
        return downloadSpeeds.isEmpty();
    }

            totalLines = lines.size();

            scrollOffset = Math.max(0, Math.min(scrollOffset, maxScroll));
            start = Math.max(0, totalLines - visibleLines - scrollOffset);
            end = Math.min(totalLines, start + visibleLines + 1);
            toDraw = new ArrayList<>(lines.subList(start, end));

    private void updateScrollIfAtBottom() {
        if (!initialized) return;
        if (isAtBottom()) scrollToBottom();
    }

    private boolean isAtBottom() {
        if (!initialized) return true;
        List<Component> lines = instanceLines.getOrDefault(currentInstanceId, Collections.emptyList());
        int visible = (textAreaBottom - textAreaTop) / lineHeight;
        if (visible <= 0) return true;
        int max = Math.max(0, lines.size() - visible);
        return scrollOffset >= max;
    }

    private void scrollToBottom() {
        if (!initialized) return;
        List<Component> lines = instanceLines.getOrDefault(currentInstanceId, Collections.emptyList());
        int visible = (textAreaBottom - textAreaTop) / lineHeight;
        scrollOffset = Math.max(0, lines.size() - visible);
    }

    @Override
    public boolean mouseClicked(double mouseX, double mouseY, int button) {
        if (button == 0 && isOverScrollbar(mouseX, mouseY)) {
            isDraggingScrollbar = true;
            updateScrollFromMouseY(mouseY);
            return true;
        }
        return super.mouseClicked(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseReleased(double mouseX, double mouseY, int button) {
        if (button == 0) isDraggingScrollbar = false;
        return super.mouseReleased(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseDragged(double mouseX, double mouseY, int button, double dragX, double dragY) {
        if (isDraggingScrollbar) {
            updateScrollFromMouseY(mouseY);
            return true;
        }
        return super.mouseDragged(mouseX, mouseY, button, dragX, dragY);
    }

    private boolean isOverScrollbar(double mouseX, double mouseY) {
        if (!initialized || textAreaBottom <= textAreaTop) return false;
        List<Component> lines = instanceLines.getOrDefault(currentInstanceId, Collections.emptyList());
        int visible = (textAreaBottom - textAreaTop) / lineHeight;
        if (visible <= 0 || lines.size() <= visible) return false;
        int barWidth = 8;
        int barX = this.width - barWidth - 4;
        return mouseX >= barX && mouseX <= barX + barWidth && mouseY >= textAreaTop && mouseY <= textAreaBottom;
    }

    private void updateScrollFromMouseY(double mouseY) {
        if (!initialized || textAreaBottom <= textAreaTop) return;
        List<Component> lines = instanceLines.getOrDefault(currentInstanceId, Collections.emptyList());
        int visible = (textAreaBottom - textAreaTop) / lineHeight;
        int totalLines = lines.size();
        if (visible <= 0 || totalLines <= visible) return;
        int maxScroll = Math.max(0, totalLines - visible);
        int trackHeight = textAreaBottom - textAreaTop;
        int barHeight = Math.max(20, (int)((float)visible / totalLines * trackHeight));
        double usable = trackHeight - barHeight;
        if (usable <= 0) return;
        double percent = Math.max(0, Math.min(1, (mouseY - textAreaTop) / usable));
        scrollOffset = (int)(percent * maxScroll);
        scrollOffset = Math.max(0, Math.min(scrollOffset, maxScroll));
    }

    @Override
    public void onClose() {
        if (Minecraft.getInstance().getConnection() != null) {
            PacketDistributor.sendToServer(new ImmersiveExitPacket(currentInstanceId));
        }
        super.onClose();
    }

    private void drawChart(GuiGraphics gui, int chartY, LinkedList<Double> data, String title, int color) {
        if (chartY + CHART_HEIGHT < 0 || chartY > this.height) return;
        int plotTop = chartY + 20, plotBottom = chartY + CHART_HEIGHT - 25;
        int plotLeft = chartLeft + 40, plotRight = chartRight - 20;

        gui.fill(chartLeft, chartY + CHART_HEIGHT - 1, chartRight, chartY + CHART_HEIGHT, 0xFFAAAAAA);

        } else {
            gui.drawString(this.font, Component.translatable("mis.status.chart_loading"), plotLeft + 10, plotTop + 40, 0xFF6666, false);

    private double calcMax(LinkedList<Double> data) {
        if (data.isEmpty()) return 100;
        double max = 0;
        for (double v : data) if (v > max) max = v;
        double ext = max * 1.2;
        int step = ext <= 200 ? 10 : 50;
        return Math.ceil(ext / step) * step;
    }

    private void drawPolyline(GuiGraphics gui, LinkedList<Double> data, int left, int right, int top, int bottom, double max, int color) {
        int n = data.size();
        int prevX = -1, prevY = -1;
        for (int i = 0; i < n; i++) {
            double val = data.get(i);
            int x = left + (right - left) * i / (n - 1);
            int y = bottom - (int) ((val / max) * (bottom - top));
            y = Math.max(top, Math.min(bottom, y));
            if (prevX != -1) drawLine(gui, prevX, prevY, x, y, color);
            gui.fill(x - 1, y - 1, x + 1, y + 1, color);
            prevX = x;
            prevY = y;
        }
    }

    private void drawLine(GuiGraphics gui, int x1, int y1, int x2, int y2, int color) {
        int dx = Math.abs(x2 - x1), dy = Math.abs(y2 - y1);
        int sx = x1 < x2 ? 1 : -1, sy = y1 < y2 ? 1 : -1;
        int err = dx - dy, x = x1, y = y1;
        while (true) {
            gui.fill(x, y, x + 1, y + 1, color);
            if (x == x2 && y == y2) break;
            int e2 = err * 2;
            if (e2 > -dy) { err -= dy; x += sx; }
            if (e2 < dx) { err += dx; y += sy; }
        }
    }

package com.xiaoze_qwq_.mis.client.util;

import net.minecraft.network.chat.Component;
import net.minecraft.network.chat.MutableComponent;
import net.minecraft.network.chat.Style;
import net.minecraft.network.chat.TextColor;

import java.util.*;

public class ANSITextProcessorClient {
    private static final int CACHE_SIZE = 200;
    private static final LinkedHashMap<String, Component> CACHE = new LinkedHashMap<String, Component>(CACHE_SIZE, 0.75f, true) {
        @Override
        protected boolean removeEldestEntry(Map.Entry<String, Component> eldest) {
            return size() > CACHE_SIZE;
        }
    };

    private static class StyleState {
        TextColor foregroundColor;
        boolean bold;
        boolean italic;
        boolean underlined;

        StyleState() { reset(); }

        void reset() {
            foregroundColor = null;
            bold = false;
            italic = false;
            underlined = false;
        }

        StyleState copy() {
            StyleState copy = new StyleState();
            copy.foregroundColor = this.foregroundColor;
            copy.bold = this.bold;
            copy.italic = this.italic;
            copy.underlined = this.underlined;
            return copy;
        }

        boolean equals(StyleState other) {
            if (other == null) return false;
            return Objects.equals(foregroundColor, other.foregroundColor) &&
                    bold == other.bold &&
                    italic == other.italic &&
                    underlined == other.underlined;
        }
    }

    private static final String[] ANSI_COLOR_HEX = {
            "#000000", "#AA0000", "#00AA00", "#AA5500",
            "#0000AA", "#AA00AA", "#00AAAA", "#AAAAAA",
            "#555555", "#FF5555", "#55FF55", "#FFFF55",
            "#5555FF", "#FF55FF", "#55FFFF", "#FFFFFF"
    };

    private static String ansiCodeToHex(int code) {
        int index;
        if (code >= 30 && code <= 37) index = code - 30;
        else if (code >= 90 && code <= 97) index = code - 90 + 8;
        else return null;
        return ANSI_COLOR_HEX[index];
    }

    private static TextColor parseColor(String hex) {
        return TextColor.parseColor(hex).result().orElse(null);
    }

    private static void applySgrParameters(StyleState current, List<Integer> params) {
        int i = 0;
        while (i < params.size()) {
            int code = params.get(i);
            if (code == 0) {
                current.reset();
            } else if (code == 1) {
                current.bold = true;
            } else if (code == 3) {
                current.italic = true;
            } else if (code == 4) {
                current.underlined = true;
            } else if (code == 22) {
                current.bold = false;
            } else if (code == 23) {
                current.italic = false;
            } else if (code == 24) {
                current.underlined = false;
            } else if (code == 39) {
                current.foregroundColor = null;
            } else if ((code >= 30 && code <= 37) || (code >= 90 && code <= 97)) {
                String hex = ansiCodeToHex(code);
                if (hex != null) {
                    current.foregroundColor = parseColor(hex);
                }
            } else if (code == 38) {
                if (i + 1 < params.size()) {
                    int mode = params.get(i + 1);
                    if (mode == 2 && i + 4 < params.size()) {
                        int r = params.get(i + 2);
                        int g = params.get(i + 3);
                        int b = params.get(i + 4);
                        String hex = String.format("#%02X%02X%02X", r, g, b);
                        current.foregroundColor = parseColor(hex);
                        i += 4;
                    } else if (mode == 5 && i + 2 < params.size()) {
                        int colorIndex = params.get(i + 2);
                        if (colorIndex >= 0 && colorIndex < 256) {
                            String hex;
                            if (colorIndex < 16) {
                                hex = ANSI_COLOR_HEX[colorIndex];
                            } else if (colorIndex < 232) {
                                int idx = colorIndex - 16;
                                int r = (idx / 36) * 51;
                                int g = ((idx / 6) % 6) * 51;
                                int b = (idx % 6) * 51;
                                hex = String.format("#%02X%02X%02X", r, g, b);
                            } else {
                                int gray = (colorIndex - 232) * 10 + 8;
                                hex = String.format("#%02X%02X%02X", gray, gray, gray);
                            }
                            current.foregroundColor = parseColor(hex);
                        }
                        i += 2;
                    }
                }
            } else if (code == 48) {

                    if (mode == 2) i += 4;

    public static Component toTextComponent(String ansiText) {
        if (ansiText == null) return Component.empty();
        synchronized (CACHE) {
            Component cached = CACHE.get(ansiText);
            if (cached != null) return cached;
        }
        Component result = parseInternal(ansiText);
        synchronized (CACHE) {
            CACHE.put(ansiText, result);
        }
        return result;
    }

    private static Component parseInternal(String ansiText) {
        List<MutableComponent> components = new ArrayList<>();
        StyleState current = new StyleState();
        StringBuilder textBuffer = new StringBuilder();
        int index = 0, len = ansiText.length();

        while (index < len) {
            char ch = ansiText.charAt(index);
            if (ch == '\u001B' && index + 1 < len && ansiText.charAt(index + 1) == '[') {
                if (textBuffer.length() > 0) {
                    components.add(createComponent(textBuffer.toString(), current));
                    textBuffer.setLength(0);
                }
                int seqStart = index + 2, cmdEnd = seqStart;
                while (cmdEnd < len && !Character.isLetter(ansiText.charAt(cmdEnd))) cmdEnd++;
                if (cmdEnd < len) {
                    String paramStr = ansiText.substring(seqStart, cmdEnd);
                    char cmd = ansiText.charAt(cmdEnd);
                    index = cmdEnd + 1;
                    if (cmd == 'm') {
                        List<Integer> params = new ArrayList<>();
                        if (!paramStr.isEmpty()) {
                            for (String part : paramStr.split(";")) {
                                if (!part.isEmpty()) {
                                    try {
                                        params.add(Integer.parseInt(part));
                                    } catch (NumberFormatException ignored) {}
                                }
                            }
                        }
                        applySgrParameters(current, params);
                    }
                } else {
                    index++;
                }
            } else {
                textBuffer.append(ch);
                index++;
            }
        }
        if (textBuffer.length() > 0) {
            components.add(createComponent(textBuffer.toString(), current));
        }

        List<MutableComponent> merged = new ArrayList<>();
        for (MutableComponent comp : components) {
            if (!merged.isEmpty() && haveSameStyle(merged.get(merged.size() - 1), comp)) {
                merged.get(merged.size() - 1).append(comp);
            } else {
                merged.add(comp);
            }
        }

        MutableComponent result = Component.empty();
        for (MutableComponent comp : merged) result.append(comp);
        return result;
    }

    private static boolean haveSameStyle(MutableComponent a, MutableComponent b) {
        Style sa = a.getStyle();
        Style sb = b.getStyle();
        return Objects.equals(sa.getColor(), sb.getColor()) &&
                sa.isBold() == sb.isBold() &&
                sa.isItalic() == sb.isItalic() &&
                sa.isUnderlined() == sb.isUnderlined();
    }

    private static MutableComponent createComponent(String text, StyleState style) {
        MutableComponent comp = Component.literal(text);
        Style s = comp.getStyle();
        if (style.foregroundColor != null) s = s.withColor(style.foregroundColor);
        if (style.bold) s = s.withBold(true);
        if (style.italic) s = s.withItalic(true);
        if (style.underlined) s = s.withUnderlined(true);
        comp.setStyle(s);
        return comp;
    }

                        src.sendFailure(Component.translatable("mis.error.verify_locked", lockRemaining / 1000));

                        src.sendFailure(Component.translatable("mis.error.wrong_password"));
                    }

        });

    private static long parseDuration(String input) {
        if (input == null || input.isEmpty()) return MISSessionManager.defaultDurationMs();
        String s = input.trim().toLowerCase(java.util.Locale.ROOT);
        try {
            if (s.endsWith("s")) return Long.parseLong(s.substring(0, s.length() - 1)) * 1000L;
            if (s.endsWith("m")) return Long.parseLong(s.substring(0, s.length() - 1)) * 60_000L;
            if (s.endsWith("h")) return Long.parseLong(s.substring(0, s.length() - 1)) * 3_600_000L;
            if (s.endsWith("d")) return Long.parseLong(s.substring(0, s.length() - 1)) * 86_400_000L;
            return Long.parseLong(s) * 1000L;
        } catch (NumberFormatException e) {
            return MISSessionManager.defaultDurationMs();
        }
    }

    private static byte[] pbkdf2(char[] password, byte[] salt, int iterations) {
        try {
            PBEKeySpec spec = new PBEKeySpec(password, salt, iterations, 512);
            SecretKeyFactory factory = SecretKeyFactory.getInstance(PBKDF2_ALGO);
            return factory.generateSecret(spec).getEncoded();
        } catch (Exception e) {
            throw new IllegalStateException("PBKDF2 not available", e);
        }
    }

    private static String hashPassword(String plain) {
        byte[] salt = new byte[SALT_BYTES];
        RNG.nextBytes(salt);
        byte[] hash = pbkdf2(plain.toCharArray(), salt, PBKDF2_ITERATIONS);
        return PBKDF2_ITERATIONS + "$" + Base64.getEncoder().encodeToString(salt)
                + "$" + Base64.getEncoder().encodeToString(hash);
    }

    private static boolean verifyStoredPassword(String plain, String stored) {
        if (stored == null || stored.isEmpty()) return false;
        try {
            String[] parts = stored.split("\\$");
            if (parts.length != 3) return false;
            int iterations = Integer.parseInt(parts[0]);
            byte[] salt = Base64.getDecoder().decode(parts[1]);
            byte[] expected = Base64.getDecoder().decode(parts[2]);
            byte[] actual = pbkdf2(plain.toCharArray(), salt, iterations);
            return MessageDigest.isEqual(expected, actual);
        } catch (Exception e) {
            return false;
        }
    }

    private static void loadPasswordHash() {

    private static String formatInlineTable(Map<String, String> env) {

public class MCCOutputLogger {
    private static final Logger LOGGER = LoggerFactory.getLogger("MCCInServer");
    private final Path logDir;
    private final MISConfig.InstanceConfig config;  // ä¿å­˜å®ä¾‹é…ç½®å¼•ç”¨
    private static final long MAX_LOG_SIZE_BYTES = 1 * 1024 * 1024;
    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.ofPattern("yyyy-MM-dd-HH-mm-ss");

    private PrintWriter currentWriter;
    private Path currentLogPath;

    public MCCOutputLogger(Path logDirectory, MISConfig.InstanceConfig config) {
        this.logDir = logDirectory;
        this.config = config;
        initLogDir();
    }

    private void initLogDir() {
        try {
            if (!Files.exists(logDir)) Files.createDirectories(logDir);
        } catch (IOException e) {
            LOGGER.error("Failed to create log directory: " + logDir, e);
        }
    }

    public synchronized void startNewLog() {
        if (currentWriter != null) currentWriter.close();
        currentStartTime = Instant.now();
        String fileName = "MCC-Output-" + DATE_FORMATTER.format(
                LocalDateTime.ofInstant(currentStartTime, ZoneId.systemDefault())) + ".log";
        currentLogPath = logDir.resolve(fileName);
        try {
            currentWriter = new PrintWriter(Files.newBufferedWriter(currentLogPath, StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE, StandardOpenOption.WRITE, StandardOpenOption.APPEND));
            LOGGER.info("[{}] Started new MCC log: {}", config.id, currentLogPath);
        } catch (IOException e) {
            LOGGER.error("[{}] Failed to create log file", config.id, e);
            currentWriter = null;
        }
    }

    public synchronized void writeLine(String line) {
        if (currentWriter == null) return;
        String outputLine = line;
        if (config.saveLogTime) {
            String timestamp = LocalDateTime.now().format(DateTimeFormatter.ofPattern(config.timeFormat));
            outputLine = "[" + timestamp + "] " + line;
        }
        currentWriter.println(outputLine);
        currentWriter.flush();
        checkSizeAndRotate();
    }

    private void checkSizeAndRotate() {
        if (currentLogPath != null && Files.exists(currentLogPath)) {
            try {
                if (Files.size(currentLogPath) >= MAX_LOG_SIZE_BYTES) {
                    LOGGER.info("[{}] Log file exceeded 1MB, rotating...", config.id);
                    rotate();
                }
            } catch (IOException e) {
                LOGGER.error("[{}] Failed to check log size", config.id, e);
            }
        }
    }

    public synchronized void rotate() {
        if (currentWriter == null) return;
        currentWriter.close();
        currentWriter = null;
        if (currentLogPath != null && Files.exists(currentLogPath)) {
            compressToTarGz(currentLogPath);
        }
        startNewLog();
    }

    private void compressToTarGz(Path logFile) {
        try {
            String baseName = logFile.getFileName().toString();
            String tarGzName = baseName.replace(".log", ".tar.gz");
            Path tarGzPath = logFile.resolveSibling(tarGzName);
            try (OutputStream fileOut = Files.newOutputStream(tarGzPath, StandardOpenOption.CREATE, StandardOpenOption.WRITE);
                 BufferedOutputStream bufferedOut = new BufferedOutputStream(fileOut);
                 GzipCompressorOutputStream gzipOut = new GzipCompressorOutputStream(bufferedOut);
                 TarArchiveOutputStream tarOut = new TarArchiveOutputStream(gzipOut)) {

                tarOut.setLongFileMode(TarArchiveOutputStream.LONGFILE_POSIX);
                TarArchiveEntry entry = new TarArchiveEntry(logFile.toFile(), baseName);
                entry.setSize(Files.size(logFile));
                tarOut.putArchiveEntry(entry);
                Files.copy(logFile, tarOut);
                tarOut.closeArchiveEntry();
                tarOut.finish();
                LOGGER.info("[{}] Compressed log to {}", config.id, tarGzPath);
                Files.delete(logFile);
            }
        } catch (IOException e) {
            LOGGER.error("[{}] Failed to compress log file: " + logFile, config.id, e);
        }
    }

    public synchronized void closeAndFinalize() {
        if (currentWriter != null) {
            currentWriter.close();
            currentWriter = null;
        }
        if (currentLogPath != null && Files.exists(currentLogPath)) {

import com.xiaoze_qwq_.mis.MCCInstance;

import java.time.Duration;
import java.time.Instant;
import java.util.List;
import java.util.concurrent.ScheduledFuture;
import java.util.concurrent.TimeUnit;

public class StatusMonitor {
    private final MCCInstance instance;
    private SystemInfo systemInfo;
    private OperatingSystem os;
    private ScheduledFuture<?> monitorTask;
    private boolean initialized = false;

    private volatile double cpuUsage = -1;
    private volatile long memoryMB = -1;
    private volatile int mccPid = -1;

    private volatile long lastTotalRx = 0;
    private volatile long lastTotalTx = 0;
    private volatile long lastTime = 0;
    private volatile double downloadSpeedKBps = 0.0;
    private volatile double uploadSpeedKBps = 0.0;
    private volatile long totalRxMB = 0;
    private volatile long totalTxMB = 0;
    private volatile Instant mccStartTime = null;

    public StatusMonitor(MCCInstance instance) {
        this.instance = instance;
        try {
            systemInfo = new SystemInfo();
            os = systemInfo.getOperatingSystem();
            initialized = true;
        } catch (Throwable t) {
            MCCInServer.LOGGER.warn("Failed to init OSHI for instance " + instance.getId(), t);
        }
    }

    public void start() {
        if (!initialized) return;
        updateNetworkStats();
        lastTime = System.currentTimeMillis();
        monitorTask = instance.getScheduler().scheduleAtFixedRate(() -> {
            if (instance.getConfig().enableProcessMonitor) updateProcessStats();
            if (instance.getConfig().enableNetworkMonitor) updateNetworkStats();
            // æ¯æ¬¡æ›´æ–°åå‘æ‰€æœ‰çŠ¶æ€è®¢é˜…è€…æ¨é€æœ€æ–°æ•°æ®
            instance.broadcastStatus();
        }, 0, instance.getConfig().statusRefreshSeconds, TimeUnit.SECONDS);
    }

    public void stop() {
        if (monitorTask != null) monitorTask.cancel(false);
    }

    private void updateProcessStats() {
        long pid = instance.getPid();
        if (pid <= 0) {
            cpuUsage = -1; memoryMB = -1; mccPid = -1;
            return;
        }
        OSProcess process = os.getProcess((int) pid);
        if (process != null) {
            cpuUsage = process.getProcessCpuLoadCumulative() * 100;
            memoryMB = process.getResidentSetSize() / (1024 * 1024);
            mccPid = (int) pid;
            if (mccStartTime == null) mccStartTime = Instant.now();
        } else {
            cpuUsage = -1; memoryMB = -1; mccPid = -1;
        }
    }

    private synchronized void updateNetworkStats() {
        try {
            List<NetworkIF> netIfs = systemInfo.getHardware().getNetworkIFs();
            long nowRx = 0, nowTx = 0;
            for (NetworkIF net : netIfs) {
                if (!net.getName().startsWith("lo") && net.getBytesRecv() > 0) {
                    nowRx += net.getBytesRecv();
                    nowTx += net.getBytesSent();
                }
            }
            long now = System.currentTimeMillis();
            long elapsedMs = now - lastTime;
            if (lastTotalRx > 0 && elapsedMs > 0) {
                downloadSpeedKBps = (nowRx - lastTotalRx) / 1024.0 / (elapsedMs / 1000.0);
                uploadSpeedKBps = (nowTx - lastTotalTx) / 1024.0 / (elapsedMs / 1000.0);
            }
            lastTotalRx = nowRx; lastTotalTx = nowTx; lastTime = now;
            totalRxMB = lastTotalRx / (1024 * 1024);
            totalTxMB = lastTotalTx / (1024 * 1024);
        } catch (Exception e) {
            MCCInServer.LOGGER.warn("Network stats update failed for " + instance.getId(), e);
        }
    }

    public double getCpuUsage() { return cpuUsage; }
    public long getMemoryMB() { return memoryMB; }
    public int getMccPid() { return mccPid; }
    public double getDownloadSpeedKBps() { return downloadSpeedKBps; }
    public double getUploadSpeedKBps() { return uploadSpeedKBps; }
    public long getTotalRxMB() { return totalRxMB; }
    public long getTotalTxMB() { return totalTxMB; }

    public String getUptimeFormatted() {
        if (mccStartTime == null || !instance.isRunning()) return "N/A";
        Duration d = Duration.between(mccStartTime, Instant.now());
        long h = d.toHours(), m = d.toMinutesPart(), s = d.toSecondsPart();
        return h > 0 ? String.format("%dh %dm %ds", h, m, s) : (m > 0 ? String.format("%dm %ds", m, s) : String.format("%ds", s));
    }

    public String getFormattedStatus() {
        StringBuilder sb = new StringBuilder();
        sb.append("=== Instance ").append(instance.getId()).append(" Status ===\n");
        sb.append("Running: ").append(instance.isRunning()).append("\n");
        if (instance.isRunning()) {
            sb.append("Uptime: ").append(getUptimeFormatted()).append("\n");
            if (instance.getConfig().enableProcessMonitor) {
                sb.append("PID: ").append(mccPid).append("\n");
                sb.append("CPU: ").append(cpuUsage >= 0 ? String.format("%.1f%%", cpuUsage) : "N/A").append("\n");
                sb.append("Memory: ").append(memoryMB >= 0 ? memoryMB + " MB" : "N/A").append("\n");
            }
            if (instance.getConfig().enableNetworkMonitor) {
                sb.append("Net: â†“ ").append(String.format("%.1f", downloadSpeedKBps)).append(" KB/s â†‘ ").append(String.format("%.1f", uploadSpeedKBps)).append(" KB/s\n");
                sb.append("Total: â†“ ").append(totalRxMB).append(" MB â†‘ ").append(totalTxMB).append(" MB\n");
            }
        }
        return sb.toString();
    }

package com.xiaoze_qwq_.mis.util;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.locks.ReentrantReadWriteLock;

public class RingBuffer<T> {
    private Object[] buffer;  // å»æ‰ final
    private volatile int writePos = 0;
    private volatile int size = 0;
    private int capacity;
    private final ReentrantReadWriteLock lock = new ReentrantReadWriteLock();

    public RingBuffer(int capacity) {
        this.capacity = capacity;
        this.buffer = new Object[capacity];
    }

    public void add(T item) {
        lock.writeLock().lock();
        try {
            buffer[writePos] = item;
            writePos = (writePos + 1) % capacity;
            if (size < capacity) size++;
        } finally {
            lock.writeLock().unlock();
        }
    }

    public List<T> getLast(int count) {
        lock.readLock().lock();
        try {
            if (count <= 0 || size == 0) return new ArrayList<>();
            int actual = Math.min(count, size);
            List<T> result = new ArrayList<>(actual);
            int start = (writePos - size + capacity) % capacity;
            for (int i = 0; i < size; i++) {
                int idx = (start + i) % capacity;
                if (i >= size - actual) {
                    @SuppressWarnings("unchecked")
                    T item = (T) buffer[idx];
                    result.add(item);
                }
            }
            return result;
        } finally {
            lock.readLock().unlock();
        }
    }

    public int size() {
        return size;
    }

    public int getCapacity() {
        return capacity;
    }

    public void resize(int newCapacity) {
        lock.writeLock().lock();
        try {
            if (newCapacity == capacity) return;
            Object[] newBuffer = new Object[newCapacity];
            int oldSize = size;
            int start = (writePos - oldSize + capacity) % capacity;
            for (int i = 0; i < oldSize; i++) {
                newBuffer[i] = buffer[(start + i) % capacity];
            }
            this.buffer = newBuffer;  // ç°åœ¨å¯ä»¥èµ‹å€¼
            this.capacity = newCapacity;
            this.size = Math.min(oldSize, newCapacity);
            this.writePos = this.size % capacity;
        } finally {
            lock.writeLock().unlock();
        }
    }

    public void clear() {
        lock.writeLock().lock();
        try {
            for (int i = 0; i < capacity; i++) buffer[i] = null;
            writePos = 0;
            size = 0;
        } finally {
            lock.writeLock().unlock();
        }
    }

    private void startInternal() {
        processLock.writeLock().lock();
        try {
            if (running.get()) {
                LOGGER.warn("[{}] MCC already running", instanceId);
                return;
            }
            killPreviousProcess();
            outputRing.clear();
            consoleLastPrintedIndex.set(0);
            if (outputLogger != null) outputLogger.closeAndFinalize();
            outputLogger = new MCCOutputLogger(Paths.get(config.logDirectory), config);
            outputLogger.startNewLog();

            String path = config.executablePath;
            String osName = System.getProperty("os.name").toLowerCase();
            if (osName.contains("win") && !path.endsWith(".exe")) path += ".exe";

            File workDir = null;
            if (!config.workingDirectory.isEmpty()) {
                workDir = new File(config.workingDirectory);
            }

            File execFile;
            if (workDir != null) {
                execFile = new File(workDir, path);
            } else {
                execFile = new File(path);
            }

            if (!execFile.exists()) {
                LOGGER.error("[{}] Executable not found: {}", instanceId, execFile.getAbsolutePath());
                return;
            }
            if (!execFile.canExecute() && !osName.contains("win")) {
                try {
                    Runtime.getRuntime().exec("chmod +x " + execFile.getAbsolutePath());
                } catch (IOException e) {
                    LOGGER.error("[{}] chmod failed", instanceId, e);
                }
            }

            ProcessBuilder pb = new ProcessBuilder(execFile.getAbsolutePath());
            pb.directory(workDir != null ? workDir : execFile.getParentFile());
            pb.redirectErrorStream(true);
            config.environment.forEach(pb.environment()::put);

            mccProcess = pb.start();
            currentPid.set(mccProcess.pid());
            processWriter = new BufferedWriter(new OutputStreamWriter(mccProcess.getOutputStream(), StandardCharsets.UTF_8));
            processInputStream = mccProcess.getInputStream();
            running.set(true);

            outputReaderExecutor = Executors.newSingleThreadExecutor(r -> {
                Thread t = new Thread(r, "MIS-" + instanceId + "-OutputReader");
                t.setDaemon(true);
                return t;
            });
            outputReaderExecutor.submit(this::readOutputLoop);

            waitForExitThread = new Thread(() -> {
                try {
                    int exitCode = mccProcess.waitFor();
                    LOGGER.info("[{}] MCC exited with code {}", instanceId, exitCode);
                    running.set(false);
                    handleProcessExit(exitCode);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }, "MIS-" + instanceId + "-WaitForExit");
            waitForExitThread.setDaemon(true);
            waitForExitThread.start();

            statusMonitor = new StatusMonitor(this);
            if (config.enableProcessMonitor || config.enableNetworkMonitor) {
                statusMonitor.start();
            }

            LOGGER.info("[{}] MCC process started: {}", instanceId, execFile.getAbsolutePath());
        } catch (IOException e) {
            LOGGER.error("[{}] Failed to start MCC", instanceId, e);
            running.set(false);
            if (outputReaderExecutor != null) {
                outputReaderExecutor.shutdownNow();
                outputReaderExecutor = null;
            }
            if (processInputStream != null) {
                try { processInputStream.close(); } catch (IOException ignored) {}
                processInputStream = null;
            }
            if (processWriter != null) {
                try { processWriter.close(); } catch (IOException ignored) {}
                processWriter = null;
            }
        } finally {
            processLock.writeLock().unlock();
        }
    }

    private void readOutputLoop() {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(processInputStream, StandardCharsets.UTF_8))) {
            String line;
            while (running.get() && (line = reader.readLine()) != null) {
                long now = System.currentTimeMillis();
                String raw = line;
                String plain = ANSITextProcessor.toPlainText(raw);
                outputRing.add(new LogEntry(raw, now));
                if (outputLogger != null) outputLogger.writeLine(plain);
                if (immersiveConsole.get()) {
                    printToConsole(raw, now);
                }
                broadcastToSubscribers(raw);
            }
        } catch (IOException e) {
            if (running.get()) LOGGER.error("[{}] Output reader error", instanceId, e);
        }
    }

    private void stopInternal(boolean sendExit, boolean broadcastRemoved) {
        processLock.writeLock().lock();
        try {
            if (!running.get() && mccProcess == null) return;
            running.set(false);

            if (processInputStream != null) {
                try { processInputStream.close(); } catch (IOException ignored) {}
                processInputStream = null;
            }
            if (outputReaderExecutor != null) {
                outputReaderExecutor.shutdownNow();
                outputReaderExecutor = null;
            }

            if (sendExit && processWriter != null && mccProcess != null && mccProcess.isAlive()) {
                try {
                    processWriter.write("/exit " + config.exitCode);
                    processWriter.newLine();
                    processWriter.flush();
                    long giveUp = System.currentTimeMillis() + 5000;
                    while (mccProcess.isAlive() && System.currentTimeMillis() < giveUp) Thread.sleep(100);
                    if (mccProcess.isAlive()) mccProcess.destroy();
                } catch (Exception e) {
                    mccProcess.destroy();
                }
            } else {
                if (mccProcess != null && mccProcess.isAlive()) {
                    mccProcess.destroy();
                }
            }
            if (mccProcess != null && mccProcess.isAlive()) {
                try { mccProcess.waitFor(2, TimeUnit.SECONDS); } catch (InterruptedException ignored) {}
                if (mccProcess.isAlive()) {
                    mccProcess.destroyForcibly();
                    try { mccProcess.waitFor(5, TimeUnit.SECONDS); } catch (InterruptedException ignored) {}
                }
            }

            if (waitForExitThread != null) {
                waitForExitThread.interrupt();
                waitForExitThread = null;
            }
            if (processWriter != null) {
                try { processWriter.close(); } catch (IOException ignored) {}
                processWriter = null;
            }

            immersiveSubscribers.clear();
            statusSubscribers.clear();
            currentPid.set(-1);
            closeLogger();
            if (statusMonitor != null) statusMonitor.stop();
            if (broadcastRemoved) sendInstanceRemovedPacket();
        } finally {
            processLock.writeLock().unlock();
        }
    }

package com.xiaoze_qwq_.mis;

import com.xiaoze_qwq_.mis.config.MISConfig;

import java.util.*;
import java.util.concurrent.*;

public class MCCManager {
    private static final MCCManager INSTANCE = new MCCManager();
    private final Map<String, MCCInstance> instances = new ConcurrentHashMap<>();

    private MCCManager() {}

    public static MCCManager getInstance() { return INSTANCE; }

    // ==================== å®ä¾‹ç®¡ç† ====================
    public void loadInstances() {
        MISConfig.getAllInstances().forEach(cfg -> {
            MCCInstance instance = new MCCInstance(cfg.id, cfg);
            instances.put(cfg.id, instance);
            if (cfg.autoStart) {
                long delay = cfg.startupDelaySeconds;
                if (delay > 0) {
                    instance.getScheduler().schedule(instance::start, delay, TimeUnit.SECONDS);
                } else {
                    instance.start();
                }
            }
        });
    }

    public void reloadInstances() {
        Map<String, MCCInstance> oldInstances = new HashMap<>(instances);
        MISConfig.load();
        Collection<MISConfig.InstanceConfig> newConfigs = MISConfig.getAllInstances();
        for (MISConfig.InstanceConfig cfg : newConfigs) {
            MCCInstance existing = oldInstances.get(cfg.id);
            if (existing != null) {
                existing.updateConfig(cfg);
                oldInstances.remove(cfg.id);
            } else {
                MCCInstance newInstance = new MCCInstance(cfg.id, cfg);
                instances.put(cfg.id, newInstance);
                if (cfg.autoStart) {
                    long delay = cfg.startupDelaySeconds;
                    if (delay > 0) {
                        newInstance.getScheduler().schedule(newInstance::start, delay, TimeUnit.SECONDS);
                    } else {
                        newInstance.start();
                    }
                }
            }
        }
        for (MCCInstance removed : oldInstances.values()) {
            instances.remove(removed.getId());
            removed.stop(true);
        }
    }

    public void shutdownAll() {
        instances.values().forEach(inst -> inst.stop(true));
        instances.clear();
    }

    public MCCInstance getInstance(String id) {
        return instances.get(id);
    }

    public Collection<MCCInstance> getAllInstances() {
        return instances.values();
    }

    public boolean hasInstance(String id) {
        return instances.containsKey(id);
    }

    /**
     * é€šè¿‡åç§°ï¼ˆä¸åŒºåˆ†å¤§å°å†™ï¼‰æŸ¥æ‰¾å®ä¾‹
     */
    public MCCInstance getInstanceByName(String name) {
        String lowerName = name.toLowerCase(Locale.ROOT);
        for (MCCInstance inst : instances.values()) {
            if (inst.getConfig().name.toLowerCase(Locale.ROOT).equals(lowerName)) {
                return inst;
            }
        }
        return null;
    }

    // ==================== ç»Ÿä¸€ç©å®¶ç®¡ç† ====================
    public void removeDisconnectedPlayer(UUID uuid) {
        instances.values().forEach(inst -> inst.removeDisconnectedPlayer(uuid));
    }

    public void start() {
        loadInstances();
    }

    public void stop() {
        shutdownAll();
    }
}

package com.xiaoze_qwq_.mis;

import com.moandjiezana.toml.Toml;
import net.minecraft.server.level.ServerPlayer;
import net.neoforged.fml.loading.FMLPaths;

import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.io.Reader;
import java.io.Writer;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collection;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public class MISSessionManager {
    private static final MISSessionManager INSTANCE = new MISSessionManager();
    private static final Path AUTH_FILE = FMLPaths.CONFIGDIR.get().resolve("mis_auth.toml");
    private static final long DEFAULT_AUTH_DURATION_MS = 24L * 60L * 60L * 1000L;

    private final Map<UUID, MISAuthCookie> cookies = new ConcurrentHashMap<>();

    private MISSessionManager() {}

    public static MISSessionManager getInstance() {
        return INSTANCE;
    }

    public void load() {
        if (!Files.exists(AUTH_FILE)) return;
        try (Reader reader = new InputStreamReader(Files.newInputStream(AUTH_FILE), StandardCharsets.UTF_8)) {
            Toml toml = new Toml().read(reader);
            List<Toml> auths = toml.getTables("auth");
            if (auths == null) return;
            for (Toml t : auths) {
                String uuidStr = t.getString("uuid");
                if (uuidStr == null) continue;
                try {
                    UUID uuid = UUID.fromString(uuidStr);
                    MISAuthCookie cookie = new MISAuthCookie(
                            uuid,
                            t.getString("username", ""),
                            t.getString("ip", ""),
                            t.getLong("grantedAt", 0L),
                            t.getLong("expiresAt", 0L),
                            t.getString("uid", ""));
                    cookies.put(uuid, cookie);
                } catch (IllegalArgumentException ignored) {
                }
            }
        } catch (IOException e) {
            MCCInServer.LOGGER.error("Failed to load mis_auth.toml", e);
        }
    }

    private synchronized void save() {
        try (Writer writer = new OutputStreamWriter(Files.newOutputStream(AUTH_FILE), StandardCharsets.UTF_8)) {
            writer.write("# MIS Authentication Cookies\n\n");
            for (MISAuthCookie c : cookies.values()) {
                writer.write("[[auth]]\n");
                writer.write("uuid = \"" + c.uuid() + "\"\n");
                writer.write("username = \"" + escape(c.username()) + "\"\n");
                writer.write("ip = \"" + escape(c.ip()) + "\"\n");
                writer.write("grantedAt = " + c.grantedAt() + "\n");
                writer.write("expiresAt = " + c.expiresAt() + "\n");
                writer.write("uid = \"" + escape(c.uid()) + "\"\n\n");
            }
        } catch (IOException e) {
            MCCInServer.LOGGER.error("Failed to save mis_auth.toml", e);
        }
    }

    private static String escape(String s) {
        return s == null ? "" : s.replace("\\", "\\\\").replace("\"", "\\\"");
    }

    // ==================== æˆæƒç®¡ç† ====================

    public void grant(ServerPlayer player, long durationMs) {
        grant(player.getUUID(), player.getGameProfile().getName(), ipOf(player), durationMs, "");
    }

    public void grant(UUID uuid, String username, String ip, long durationMs, String uid) {
        long now = System.currentTimeMillis();
        cookies.put(uuid, new MISAuthCookie(uuid, username, ip, now, now + durationMs, uid));
        save();
    }

    public void revoke(UUID uuid) {
        cookies.remove(uuid);
        save();
    }

    public boolean isInMISMode(ServerPlayer player) {
        MISAuthCookie c = cookies.get(player.getUUID());
        if (c == null) return false;
        return c.expiresAt() > System.currentTimeMillis();
    }

    public boolean hasCookie(UUID uuid) {
        return cookies.containsKey(uuid);
    }

    public MISAuthCookie getCookie(UUID uuid) {
        return cookies.get(uuid);
    }

    public Collection<MISAuthCookie> listCookies() {
        return new ArrayList<>(cookies.values());
    }

    public void purgeExpired() {
        long now = System.currentTimeMillis();
        boolean changed = cookies.values().removeIf(c -> c.expiresAt() <= now);
        if (changed) save();
    }

    private static String ipOf(ServerPlayer player) {
        try {
            if (player.connection != null && player.connection.getRemoteAddress() != null) {
                return player.connection.getRemoteAddress().toString();
            }
        } catch (Exception ignored) {
        }
        return "";
    }

    public static long defaultDurationMs() {
        return DEFAULT_AUTH_DURATION_MS;
    }

    public record MISAuthCookie(UUID uuid, String username, String ip, long grantedAt, long expiresAt, String uid) {
        public long remainingMs() {
            return expiresAt - System.currentTimeMillis();
        }
    }
}
